## APPENDIX — THE MEASURING CODE, SO THE NUMBERS ARE RE-CHECKABLE

⭐ **THE EVIDENCE IS THE ARTIFACT, NEVER THE AGENT.** *"I counted it, it's in my context"* is the sound a
gate makes while it fails. Save the block below as `measure_pair04.py`, point it at this file, and it
re-derives every figure in the ledger above — style/exclude/lyrics character counts, sung-line counts,
BPM, header validity, SFX-cue counts, byte-identical chorus blocks, style-field negations, Source-2
exactness **and** mid-sentence placement, word-boundary sweeps for the absolute exclusions and for named
breaks, dedication length, and within-pair token-Jaccard. **It reads the finished packages, not the
drafts, and it takes no argument from me.**

```python
#!/usr/bin/env python3
"""Measure PAIR 04 step-10 packages by parsing the finished file off disk."""
import re, sys, statistics, itertools, pathlib

P = pathlib.Path(r"E:\git\private-claw\output\daily\2026-08-18-canonical\pair_04_step10_revision_synthesis.md")
text = P.read_text(encoding="utf-8")

# --- split songs on canonical headings "# V1 — TITLE" (not ## / ###) ---
song_re = re.compile(r"^# (V\d) — (.+)$", re.M)
marks = [(m.start(), m.group(1), m.group(2)) for m in song_re.finditer(text)]
ded_start = text.index("\n## DEDICATION")
songs = []
for i, (pos, vid, title) in enumerate(marks):
    end = marks[i + 1][0] if i + 1 < len(marks) else ded_start
    songs.append((vid, title, text[pos:end]))

def field(block, head, nexts):
    s = block.index("### " + head) + len("### " + head)
    ends = [block.index("### " + n) for n in nexts if "### " + n in block]
    e = min(ends) if ends else len(block)
    return block[s:e].strip()

SOURCE2 = [
    "brightly-colored production",
    "lush, aerodynamic beats that instantly hits cruising altitude",
    "a choppier flip of the buttery soul sample to a new drum pattern",
    "the distant rollerdisco claps",
    "a two-beer chill",
]
NEG = [r"\bno\b", r"\bnot\b", r"\bnever\b", r"\bwithout\b", r"\bnone\b", r"\bnor\b",
       r"\bavoid\b", r"\babsent\b", r"\bfree of\b", r"\bun[a-z]+ed\b", r"\b\w+less\b", r"\bexcept\b"]
BANNED = ["boat", "ferry", "capsiz", "rescue", "lake", "drown", "missing", "body", "disaster",
          "kettle", "toast", "tea", "chip shop", "kitchen", "armchair", "soup", "mug", "brew"]
BREAKS = ["amen", "apache", "funky drummer", "think break", "hot pants"]

print("=" * 96)
rows, all_lyr, all_sty = [], {}, {}
for vid, title, block in songs:
    style = field(block, "SUNO STYLE PROMPT", ["SUNO EXCLUDE", "SUNO LYRICS"])
    excl = field(block, "SUNO EXCLUDE", ["SUNO LYRICS"])
    lyr = field(block, "SUNO LYRICS", [])
    lyr = lyr.rstrip("-\n ").rstrip()
    all_lyr[vid], all_sty[vid] = lyr, style

    lines = [l.strip() for l in lyr.splitlines() if l.strip()]
    heads = [l for l in lines if l.startswith("[") and l.endswith("]")]
    cues = [l for l in lines if l.startswith("*") and l.endswith("*")]
    sung = [l for l in lines if l not in heads and l not in cues]
    wpl = [len(l.split()) for l in sung]
    bpm = re.search(r"(\d{2,3})\s*BPM", style)

    bad_head = []
    for h in heads:
        parts = h[1:-1].split(" - ")
        if len(parts) != 3 or not parts[1].startswith("EMO:") or len(parts[1]) < 8:
            bad_head.append(h)
    negs = sorted({m.group(0) for p in NEG for m in re.finditer(p, style, re.I)})
    s2 = [p for p in SOURCE2 if p in style]
    s2_mid = [p for p in s2 if not re.search(r"(^|[.!?]\s+)" + re.escape(p), style)]
    banned_hits = sorted({b for b in BANNED
                          for _ in re.finditer(r"\b" + re.escape(b) + r"\b", (style + " " + lyr).lower())})
    brk = sorted({b for b in BREAKS if re.search(r"\b" + re.escape(b) + r"\b", (style + " " + excl).lower())})

    # byte-identical chorus check
    blocks, cur = {}, None
    for l in lines:
        if l in heads:
            cur = l.split(" - ")[0].lstrip("[").strip()
            blocks.setdefault(cur, []).append([])
        elif cur and l not in cues:
            blocks[cur][-1].append(l)
    chor = [tuple(b) for k, v in blocks.items() if k in ("Chorus", "Final Chorus") for b in v]
    chor_ok = len(set(chor)) == 1 and len(chor) == 4

    rows.append((vid, title, len(style), len(excl), len(lyr), len(sung), len(heads), len(cues),
                 bpm.group(1) if bpm else "??", len(s2), len(s2_mid), negs, bad_head,
                 banned_hits, brk, chor_ok, statistics.median(wpl), max(wpl), max(len(l.split()) for l in sung if len(l.split()) < 11)))

hdr = f"{'':4} {'style':>6} {'excl':>5} {'lyrics':>7} {'sung':>5} {'hdrs':>5} {'cues':>5} {'BPM':>4} {'S2':>3} {'mid':>4} {'chorus':>7} {'medWPL':>7} {'maxWPL':>7} {'nonRad':>7}"
print(hdr)
for r in rows:
    print(f"{r[0]:4} {r[2]:>6} {r[3]:>5} {r[4]:>7} {r[5]:>5} {r[6]:>5} {r[7]:>5} {r[8]:>4} {r[9]:>3} {r[10]:>4} "
          f"{'4x OK' if r[15] else 'FAIL':>7} {r[16]:>7} {r[17]:>7} {r[18]:>7}   {r[1]}")

print("\n--- violations ---")
for r in rows:
    if r[11]: print(f"  {r[0]} STYLE NEGATIONS: {r[11]}")
    if r[12]: print(f"  {r[0]} BAD EMO HEADERS: {r[12]}")
    if r[13]: print(f"  {r[0]} BANNED VOCAB (word-boundary): {r[13]}")
    if r[14]: print(f"  {r[0]} NAMED BREAK: {r[14]}")
    if r[2] > 1000: print(f"  {r[0]} STYLE OVER 1000: {r[2]}")
    if r[4] >= 5000: print(f"  {r[0]} LYRICS OVER 5000: {r[4]}")
    if not (70 <= r[5] <= 120): print(f"  {r[0]} SUNG LINES OUT OF BAND: {r[5]}")
    if r[9] < 2: print(f"  {r[0]} SOURCE-2 FLOOR MISSED: {r[9]}")
    if r[10] != r[9]: print(f"  {r[0]} SOURCE-2 NOT MID-SENTENCE: {r[9]-r[10]}")
print("  (empty above = clean)")

# substring 'body' audit
low = " ".join(all_lyr.values()).lower()
print(f"\n'body' substring count: {low.count('body')}  |  standalone \\bbody\\b: {len(re.findall(r'\\bbody\\b', low))}")

# dedication
d = text[ded_start:]
para = [p.strip() for p in d.split("\n\n") if p.strip() and not p.strip().startswith("##")][0]
print(f"DEDICATION chars: {len(para)}  (ceiling 500)")

# within-pair similarity (token Jaccard)
def toks(s): return set(re.findall(r"[a-z']+", s.lower()))
print("\n--- within-pair Jaccard (lyrics / style) ---")
for a, b in itertools.combinations(sorted(all_lyr), 2):
    jl = len(toks(all_lyr[a]) & toks(all_lyr[b])) / len(toks(all_lyr[a]) | toks(all_lyr[b]))
    js = len(toks(all_sty[a]) & toks(all_sty[b])) / len(toks(all_sty[a]) | toks(all_sty[b]))
    print(f"  {a}-{b}  lyrics {jl:.3f}   style {js:.3f}")

# closing lines
print("\n--- closing sung lines (return-swap material) ---")
for vid, title, block in songs:
    lines = [l.strip() for l in all_lyr[vid].splitlines() if l.strip()]
    sung = [l for l in lines if not (l.startswith("[") or l.startswith("*"))]
    print(f"  {vid}: ...{' / '.join(sung[-3:])}")
```
